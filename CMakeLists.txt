cmake_minimum_required(VERSION 3.16)
project(luca VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LUA REQUIRED IMPORTED_TARGET lua5.4)
pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl)
add_library(luca_core src/irc.c src/config.c src/transport.c src/sasl.c)
target_include_directories(luca_core PUBLIC include)
target_link_libraries(luca_core PUBLIC PkgConfig::OPENSSL)
target_compile_options(luca_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(luca src/main.c src/lua_runtime.c)
target_link_libraries(luca PRIVATE luca_core PkgConfig::LUA)
target_include_directories(luca PRIVATE include)
enable_testing()
add_executable(test_irc tests/test_irc.c)
target_link_libraries(test_irc PRIVATE luca_core)
add_test(NAME irc_parser COMMAND test_irc)
add_executable(test_sasl tests/test_sasl.c)
target_link_libraries(test_sasl PRIVATE luca_core)
add_test(NAME sasl COMMAND test_sasl)
