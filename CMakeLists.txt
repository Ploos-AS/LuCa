cmake_minimum_required(VERSION 3.16)
project(luca VERSION 0.1.2 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LUA REQUIRED IMPORTED_TARGET lua5.4)
pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl)
pkg_check_modules(CURL IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
add_library(luca_core src/irc.c src/config.c src/transport.c src/sasl.c src/pbmp.c src/command_queue.c src/log.c src/botai.c src/botai_scopes.c)
target_include_directories(luca_core PUBLIC include)
target_link_libraries(luca_core PUBLIC PkgConfig::OPENSSL Threads::Threads)
if(CURL_FOUND)
  target_link_libraries(luca_core PUBLIC PkgConfig::CURL)
  target_compile_definitions(luca_core PRIVATE LUCA_HAVE_CURL=1)
endif()
target_compile_options(luca_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(luca src/main.c src/lua_runtime.c)
target_link_libraries(luca PRIVATE luca_core PkgConfig::LUA)
target_include_directories(luca PRIVATE include)
enable_testing()
add_executable(test_irc tests/test_irc.c)
target_link_libraries(test_irc PRIVATE luca_core)
add_test(NAME irc_parser COMMAND test_irc)
add_executable(test_sasl tests/test_sasl.c)
target_link_libraries(test_sasl PRIVATE luca_core)
add_test(NAME sasl COMMAND test_sasl)
add_executable(test_pbmp tests/test_pbmp.c)
target_link_libraries(test_pbmp PRIVATE luca_core)
add_test(NAME pbmp COMMAND test_pbmp)

add_executable(test_botai tests/test_botai.c)
target_link_libraries(test_botai PRIVATE luca_core)
add_test(NAME botai_history COMMAND test_botai)

add_executable(test_botai_scopes tests/test_botai_scopes.c)
target_link_libraries(test_botai_scopes PRIVATE luca_core)
add_test(NAME botai_scopes COMMAND test_botai_scopes)
